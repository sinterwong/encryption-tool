CMAKE_MINIMUM_REQUIRED(VERSION 3.10)

PROJECT(encrypt)

SET(CMAKE_POSITION_INDEPENDENT_CODE ON) 

SET(DEPENDENCY_INCLUDES
    ${PROJECT_INCLUDE_DIR}
    ${PROJECT_INCLUDE_DIR}/encrypt
)

SET(DEPENDENCY_LIBS
	cryptopp::cryptopp
)

FILE(GLOB CURRENT_DIR_HEAD ${CMAKE_CURRENT_SOURCE_DIR}/crypto.hpp)
FILE(GLOB CURRENT_DIR_SRCS ${CMAKE_CURRENT_SOURCE_DIR}/*.cpp)

FILE(MAKE_DIRECTORY ${PROJECT_INCLUDE_DIR}/encrypt)
FOREACH(include ${CURRENT_DIR_HEAD})
	MESSAGE("-- Copying ${include}")
	CONFIGURE_FILE(${include} ${PROJECT_INCLUDE_DIR}/encrypt COPYONLY)
ENDFOREACH()

ADD_LIBRARY(encrypt SHARED ${CURRENT_DIR_SRCS})
TARGET_INCLUDE_DIRECTORIES(encrypt PRIVATE ${DEPENDENCY_INCLUDES})
TARGET_LINK_LIBRARIES(encrypt PRIVATE ${DEPENDENCY_LIBS})
